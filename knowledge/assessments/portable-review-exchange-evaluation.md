Reproduce archive integrity from the checkout without extracting it:

```python
import gzip
import hashlib
import json
import tarfile
from pathlib import Path

root = Path("knowledge/assessments/review-exchange-evidence")
index = json.loads((root / "index.json").read_text())
archive = root / index["archive"]
assert archive.stat().st_size == index["bytes"]
assert hashlib.sha256(archive.read_bytes()).hexdigest() == index["sha256"]
packed = (root / index["member_index"]["file"]).read_bytes()
assert hashlib.sha256(packed).hexdigest() == index["member_index"]["sha256"]
raw = gzip.decompress(packed)
assert hashlib.sha256(raw).hexdigest() == index["member_index"]["uncompressed_sha256"]
expected = json.loads(raw)
with tarfile.open(archive, "r:gz") as retained:
    members = retained.getmembers()
    assert len(members) == len(expected) == index["member_index"]["entries"]
    assert {m.name for m in members} == set(expected)
    for member in members:
        assert member.isfile()
        stream = retained.extractfile(member)
        assert stream is not None
        data = stream.read()
        assert len(data) == expected[member.name]["bytes"]
        assert hashlib.sha256(data).hexdigest() == expected[member.name]["sha256"]
print("All original archive members verified")
```
